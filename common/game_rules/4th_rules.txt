# this = country
# from = location (galactic_object)
can_explore_rifts_outside_borders = {
	exists = this
	exists = from
	OR = {
		is_nomadic = yes
		has_origin = origin_riftworld
		AND = {
			is_ai = no
			has_tradition = tr_archivism_frontier_archaeology
		}
	}
	OR = {
		from = { # regular case
			solar_system = { NOT = { exists = owner } }
		}
		AND = { # Nomadic with Research Agreement
			is_nomadic = yes
			from = {
				solar_system = { exists = owner }
			}
			is_nomadic = yes
			has_research_agreement = from.solar_system.owner
		}
		AND = {
			is_ai = no
			has_tradition = tr_archivism_frontier_archaeology
		}
	}
}
